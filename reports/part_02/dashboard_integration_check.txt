import { useEffect, useState } from "react";
import App from "./App.jsx";
import AuthScreen from "./auth/AuthScreen.jsx";
import RolePortal from "./portals/RolePortal.jsx";

import { API_BASE_URL as API } from "./services/api.js";

export default function AppRouter() {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [workspace, setWorkspace] = useState(false);

  useEffect(() => {
    let active = true;

    async function verifySession() {
      const token = localStorage.getItem("urban_ripple_token");

      if (!token) {
        if (active) setLoading(false);
        return;
      }

      try {
        const response = await fetch(`${API}/api/auth/me`, {
          headers: { Authorization: `Bearer ${token}` }
        });

        if (!response.ok) throw new Error("Session expired");

        const data = await response.json();
        if (active) setUser(data.user);
      } catch {
        localStorage.removeItem("urban_ripple_token");
        if (active) setUser(null);
      } finally {
        if (active) setLoading(false);
      }
    }

    verifySession();
    return () => { active = false; };
  }, []);

  function logout() {
    localStorage.removeItem("urban_ripple_token");
    setUser(null);
    setWorkspace(false);
  }

  if (loading) {
    return <div style={{ padding: 40, fontFamily: "Arial" }}>Verifying session...</div>;
  }

  if (!user) {
    return <AuthScreen onLogin={setUser} />;
  }

  if (!workspace) {
    return (
      <RolePortal
        user={user}
        onOpenWorkspace={() => setWorkspace(true)}
        onLogout={logout}
      />
    );
  }

  return (
    <div>
      <div style={{
        padding: "10px 20px",
        background: "#f1f1fa",
        display: "flex",
        justifyContent: "space-between",
        alignItems: "center"
      }}>
        <span>Signed in: {user.name} ({user.role})</span>
        <button onClick={() => setWorkspace(false)}>Portal</button>
        <button onClick={logout}>Logout</button>
      </div>
      <App />
    </div>
  );
}

def require_roles(*allowed_roles):
    def dependency(user=Depends(get_current_user)):
        role = str(user.get("role", "public")).lower()

        if role not in allowed_roles:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Your role is not permitted to perform this action."
            )

        return user

    return dependency


def require_permission(permission):
    def dependency(user=Depends(get_current_user)):
        role = str(user.get("role", "public")).lower()

            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"Permission denied: {permission}"
            )

        return user

    return dependency

import {
  ResponsiveContainer,
  BarChart,
  Bar,
  XAxis,
  YAxis,
  CartesianGrid,
  Tooltip,
  PieChart,
  Pie,
  Cell,
  Legend,
  LineChart,
  Line
} from "recharts";

import "./AnalyticsCharts.css";

const COLORS = [
  "#2878d0",
  "#16a085",
  "#e6a23c",
  "#e05252",
  "#805ad5",
  "#64748b"
];

export default function AnalyticsCharts({ data }) {
  if (!data) return null;

  const severityData = data.severity_distribution || [];
  const typeData = data.disruption_types || [];
  const monthlyData = data.monthly_trend || [];

  const locationData = [...(data.location_impact || [])]
    .sort(
      (a, b) =>
        Number(b.affected_population || 0) -
        Number(a.affected_population || 0)
    )
    .slice(0, 10);

  return (
    <section className="urban-charts">
      <article className="urban-chart-card">
        <h2>Monthly Disruption Activity</h2>
        <ResponsiveContainer width="100%" height={280}>
          <LineChart data={monthlyData}>
            <CartesianGrid strokeDasharray="3 3" />
            <XAxis dataKey="month" />
            <YAxis allowDecimals={false} />
            <Tooltip />
            <Line
              type="monotone"
              dataKey="count"
              name="Disruptions"
              stroke="#2878d0"
              strokeWidth={3}
              activeDot={{ r: 7 }}
            />
          </LineChart>
        </ResponsiveContainer>
      </article>

      <article className="urban-chart-card">
        <h2>Severity Distribution</h2>
        <ResponsiveContainer width="100%" height={280}>
          <PieChart>
            <Pie
              data={severityData}
              dataKey="count"
              nameKey="severity"
              cx="50%"
              cy="50%"
              outerRadius={90}
              label
            >
              {severityData.map((item, index) => (
                <Cell
                  key={`${item.severity}-${index}`}
                  fill={COLORS[index % COLORS.length]}
                />
              ))}
            </Pie>
            <Tooltip />
            <Legend />
          </PieChart>
        </ResponsiveContainer>
      </article>

      <article className="urban-chart-card">
        <h2>Disruption Categories</h2>
        <ResponsiveContainer width="100%" height={300}>
          <BarChart data={typeData} layout="vertical">
            <CartesianGrid strokeDasharray="3 3" />
            <XAxis type="number" allowDecimals={false} />
            <YAxis
              type="category"
              dataKey="disruption_type"
              width={120}
            />
            <Tooltip />
            <Bar
              dataKey="count"
              name="Disruptions"
              fill="#16a085"
              radius={[0, 6, 6, 0]}
            />
          </BarChart>
        </ResponsiveContainer>
      </article>

      <article className="urban-chart-card">
        <h2>Population Affected by Location</h2>
        {locationData.length === 0 ? (
          <p>No location impact data available.</p>
        ) : (
          <ResponsiveContainer width="100%" height={300}>
            <BarChart data={locationData} layout="vertical">
              <CartesianGrid strokeDasharray="3 3" />
              <XAxis type="number" />
              <YAxis
                type="category"
                dataKey="location"
                width={120}
              />
              <Tooltip />
              <Bar
                dataKey="affected_population"
                name="Affected Population"
                fill="#805ad5"
                radius={[0, 6, 6, 0]}
              />
            </BarChart>
          </ResponsiveContainer>
        )}
      </article>
    </section>
  );
}
