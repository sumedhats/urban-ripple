function MapAutoFit({ events }) {
  const map = useMap();

  useEffect(() => {
    const locations = events
      .filter(event => event.latitude != null && event.longitude != null)
      .map(event => [Number(event.latitude), Number(event.longitude)]);

    if (locations.length === 1) {
      map.setView(locations[0], 14);
    } else if (locations.length > 1) {
      map.fitBounds(locations, { padding: [40, 40], maxZoom: 14 });
    }
  }, [events, map]);

  return null;
}

import { useEffect, useState } from "react";
import App from "./App.jsx";
import AuthScreen from "./auth/AuthScreen.jsx";
import RolePortal from "./portals/RolePortal.jsx";

const API = import.meta.env.VITE_API_BASE_URL || "http://127.0.0.1:8000";

export default function AppRouter() {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [workspace, setWorkspace] = useState(false);

  useEffect(() => {
    let active = true;

    async function verifySession() {
      const token = localStorage.getItem("urban_ripple_token");

      if (!token) {
        if (active) setLoading(false);
        return;
      }

      try {
        const response = await fetch(`${API}/api/auth/me`, {
          headers: { Authorization: `Bearer ${token}` }
        });

        if (!response.ok) throw new Error("Session expired");

        const data = await response.json();
        if (active) setUser(data.user);
      } catch {
        localStorage.removeItem("urban_ripple_token");
        if (active) setUser(null);
      } finally {
        if (active) setLoading(false);
      }
    }

    verifySession();
    return () => { active = false; };
  }, []);

  function logout() {
    localStorage.removeItem("urban_ripple_token");
    setUser(null);
    setWorkspace(false);
  }

  if (loading) {
    return <div style={{ padding: 40, fontFamily: "Arial" }}>Verifying session...</div>;
  }

  if (!user) {
    return <AuthScreen onLogin={setUser} />;
  }

  if (!workspace) {
    return (
      <RolePortal
        user={user}
        onOpenWorkspace={() => setWorkspace(true)}
        onLogout={logout}
      />
    );
  }

  return (
    <div>
      <div style={{
        padding: "10px 20px",
        background: "#f1f1fa",
        display: "flex",
        justifyContent: "space-between",
        alignItems: "center"
      }}>
        <span>Signed in: {user.name} ({user.role})</span>
        <button onClick={() => setWorkspace(false)}>Portal</button>
        <button onClick={logout}>Logout</button>
      </div>
      <App />
    </div>
  );
}

import { useState } from "react";

export default function AuthScreen({ onLogin }) {
  const [mode, setMode] = useState("login");
  const [name, setName] = useState("");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);

  async function submit(e) {
    e.preventDefault();
    setError("");
    setLoading(true);

    try {
      if (mode === "register") {
        const registration = await fetch(`${API}/api/auth/register`, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ name, email, password })
        });

        const result = await registration.json();

        if (!registration.ok) {
          throw new Error(result.detail || "Registration failed");
        }
      }

      const response = await fetch(`${API}/api/auth/login`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ email, password })
      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(data.detail || "Login failed");
      }

      localStorage.setItem("urban_ripple_token", data.access_token);
      onLogin(data.user);
    } catch (err) {
      setError(err.message || "Unable to connect to the server");
    } finally {
      setLoading(false);
    }
  }

  return (
    <main style={styles.page}>
      <section style={styles.card}>
        <div style={styles.logo}>UR</div>
        <h1>URBAN RIPPLE</h1>
        <p style={styles.subtitle}>City Disruption Intelligence Platform</p>

                style={styles.input}
                value={name}
                onChange={e => setName(e.target.value)}

                minLength={2}

            Password
            <input
              style={styles.input}
              type="password"
              value={password}
              onChange={e => setPassword(e.target.value)}
              required
              minLength={8}
            />
          </label>

          {error && <p style={styles.error}>{error}</p>}

          <button style={styles.button} disabled={loading}>
            {loading ? "Please wait..." :
              mode === "login" ? "Sign in" : "Create account"}
          </button>

export async function getAnalyticsOverview(startDate = "", endDate = "") {
  const token = localStorage.getItem("urban_ripple_token");

  const params = new URLSearchParams();

  if (startDate) params.set("start_date", startDate);
  if (endDate) params.set("end_date", endDate);

  const query = params.toString();
  const url = `${API_URL}/api/analytics/overview${query ? `?${query}` : ""}`;

  const response = await fetch(url, {
    headers: {
      Authorization: `Bearer ${token}`
    }
  });

  if (!response.ok) {
    const result = await response.json().catch(() => ({}));
    throw new Error(result.detail || `Analytics request failed: ${response.status}`);
  }

  return response.json();
}

@app.post("/api/impact/analyze")
def analyze_impact(data: ImpactRequest):
    # Transparent demonstration formula, not a trained prediction model.
    severity_weights = {
        "Low": 1,
        "Medium": 2,
        "High": 3,
        "Critical": 4
    }

    type_weights = {
        "Road Closure": 1.0,
        "Water Outage": 1.3,
        "Power Outage": 1.5
    }

    raw_score = (
        data.affected_population
        * min(data.duration_hours, 72)
        * severity_weights[data.severity]
        * type_weights[data.disruption_type]
    )

    impact_score = round(
        min(100, raw_score / 10000 * 100), 1
    )

    if impact_score < 25:
        impact_level = "Low"
    elif impact_score < 50:
        impact_level = "Moderate"
    elif impact_score < 75:
        impact_level = "High"
    else:
        impact_level = "Severe"

    recommendations = []

    if data.severity in ["High", "Critical"]:
        recommendations.append(
            "Prioritise the disruption for immediate review by city authorities."
        )

    if data.affected_population >= 1000:
        recommendations.append(
            "Notify affected residents and coordinate alternative services."
        )

    if data.duration_hours >= 4:
        recommendations.append(
            "Prepare a continuity plan and provide periodic public updates."
        )

    if data.disruption_type == "Road Closure":
        recommendations.append(
            "Identify alternate routes and coordinate traffic management."
        )
    elif data.disruption_type == "Water Outage":
        recommendations.append(
            "Assess backup water supply and prioritise essential facilities."
        )
    elif data.disruption_type == "Power Outage":
        recommendations.append(
            "Prioritise critical infrastructure and assess backup power."
        )

    if not recommendations:
        recommendations.append(
            "Monitor the disruption and review conditions periodically."
        )

    return {
        "impact_score": impact_score,
        "impact_level": impact_level,
        "disruption_type": data.disruption_type,
        "duration_hours": data.duration_hours,
        "affected_population": data.affected_population,
        "severity": data.severity,
        "recommendations": recommendations,
        "method": "Demonstration rule-based formula",
        "is_sap_rpt_prediction": False,
        "notice": (
            "This is a preliminary demonstration estimate, "
            "not a validated prediction or SAP-RPT output."
        )
    }

def create_token(user_id: int, email: str, role: str):
    expiry = datetime.now(timezone.utc) + timedelta(hours=8)
    payload = {
        "sub": str(user_id),
        "email": email,
        "role": role,
        "exp": expiry
    }
    return jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)


def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(security)
):
    try:
        payload = jwt.decode(
            credentials.credentials,
            SECRET_KEY,
            algorithms=[ALGORITHM]
        )
        user_id = int(payload["sub"])
    except (jwt.InvalidTokenError, KeyError, ValueError):
        raise HTTPException(status_code=401, detail="Invalid or expired token")

    with get_connection() as conn:
        user = conn.execute(
            "SELECT id, name, email, role FROM users WHERE id = ?",
            (user_id,)
        ).fetchone()

    if user is None:
        raise HTTPException(status_code=401, detail="User not found")

    return dict(user)

import sqlite3
from pathlib import Path

def get_analytics(start_date=None, end_date=None):
    conditions = []
    params = []

    if start_date:
        conditions.append("date(created_at) >= date(?)")
        params.append(start_date)

    if end_date:
        conditions.append("date(created_at) <= date(?)")
        params.append(end_date)

    where_clause = (
        " WHERE " + " AND ".join(conditions)
        if conditions else ""
    )

    with get_connection() as conn:
        total = conn.execute(
            f"SELECT COUNT(*) FROM disruptions{where_clause}",
            params
        ).fetchone()[0]

        severity = conn.execute(
            f"""
            SELECT severity, COUNT(*) AS count
            FROM disruptions
            {where_clause}
            GROUP BY severity
            """,
            params
        ).fetchall()

        types = conn.execute(
            f"""
            SELECT disruption_type, COUNT(*) AS count
            FROM disruptions
            {where_clause}
            GROUP BY disruption_type
            ORDER BY count DESC
            """,
            params
        ).fetchall()

        locations = conn.execute(
            f"""
            SELECT location, COUNT(*) AS count,
                   SUM(affected_population) AS affected_population
            FROM disruptions
            {where_clause}
            GROUP BY location
            ORDER BY affected_population DESC
            """,
            params
        ).fetchall()

        population = conn.execute(
            f"""
            SELECT COALESCE(SUM(affected_population), 0)
            FROM disruptions
            {where_clause}
            """,
            params
        ).fetchone()[0]


        risk_summary = conn.execute(
            f"""
            SELECT
                SUM(CASE WHEN lower(severity) = 'critical' THEN 1 ELSE 0 END) AS critical,
                SUM(CASE WHEN lower(severity) = 'high' THEN 1 ELSE 0 END) AS high,
                SUM(CASE WHEN lower(severity) IN ('high', 'critical') THEN 1 ELSE 0 END) AS high_risk,
                SUM(CASE WHEN lower(severity) IN ('high', 'critical')
                    THEN COALESCE(affected_population, 0) ELSE 0 END) AS high_risk_population
            FROM disruptions
            {where_clause}
            """,
            params
        ).fetchone()

        monthly = conn.execute(
            f"""
            SELECT strftime('%Y-%m', created_at) AS month,
                   COUNT(*) AS count
            FROM disruptions
            {where_clause}
            GROUP BY month
            ORDER BY month
            """,
            params
        ).fetchall()

    return {
        "total_disruptions": total,
        "affected_population": population,
        "severity_distribution": [dict(row) for row in severity],
        "disruption_types": [dict(row) for row in types],
        "location_impact": [dict(row) for row in locations],
        "monthly_trend": [dict(row) for row in monthly],
        "risk_summary": dict(risk_summary),
        "filters": {
            "start_date": start_date,
            "end_date": end_date
        }
    }

import { useEffect, useState } from "react";
import App from "./App.jsx";
import AuthScreen from "./auth/AuthScreen.jsx";
import RolePortal from "./portals/RolePortal.jsx";

const API = import.meta.env.VITE_API_BASE_URL || "http://127.0.0.1:8000";

export default function AppRouter() {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [workspace, setWorkspace] = useState(false);

  useEffect(() => {
    let active = true;

    async function verifySession() {
      const token = localStorage.getItem("urban_ripple_token");

      if (!token) {
        if (active) setLoading(false);
        return;
      }

      try {
        const response = await fetch(`${API}/api/auth/me`, {
          headers: { Authorization: `Bearer ${token}` }
        });

        if (!response.ok) throw new Error("Session expired");

        const data = await response.json();
        if (active) setUser(data.user);
      } catch {
        localStorage.removeItem("urban_ripple_token");
        if (active) setUser(null);
      } finally {
        if (active) setLoading(false);
      }
    }

    verifySession();
    return () => { active = false; };
  }, []);

  function logout() {
    localStorage.removeItem("urban_ripple_token");
    setUser(null);
    setWorkspace(false);
  }

  if (loading) {
    return <div style={{ padding: 40, fontFamily: "Arial" }}>Verifying session...</div>;
  }

  if (!user) {
    return <AuthScreen onLogin={setUser} />;
  }

  if (!workspace) {
    return (
      <RolePortal
        user={user}
        onOpenWorkspace={() => setWorkspace(true)}
        onLogout={logout}
      />
    );
  }

  return (
    <div>
      <div style={{
        padding: "10px 20px",
        background: "#f1f1fa",
        display: "flex",
        justifyContent: "space-between",
        alignItems: "center"
      }}>
        <span>Signed in: {user.name} ({user.role})</span>
        <button onClick={() => setWorkspace(false)}>Portal</button>
        <button onClick={logout}>Logout</button>
      </div>
      <App />
    </div>
  );
}
