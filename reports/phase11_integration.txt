const API = import.meta.env.VITE_API_BASE_URL || "http://127.0.0.1:8000";

export default function AppRouter() {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [workspace, setWorkspace] = useState(false);

  useEffect(() => {
    let active = true;

    async function verifySession() {
      const token = localStorage.getItem("urban_ripple_token");

      if (!token) {
        if (active) setLoading(false);
        return;
      }

      try {
        const response = await fetch(`${API}/api/auth/me`, {
          headers: { Authorization: `Bearer ${token}` }
        });

        if (!response.ok) throw new Error("Session expired");

        const data = await response.json();
        if (active) setUser(data.user);
      } catch {
        localStorage.removeItem("urban_ripple_token");
        if (active) setUser(null);
      } finally {
        if (active) setLoading(false);
      }
    }

    verifySession();
    return () => { active = false; };
  }, []);

  function logout() {
    localStorage.removeItem("urban_ripple_token");
    setUser(null);
    setWorkspace(false);
  }

  if (loading) {
    return <div style={{ padding: 40, fontFamily: "Arial" }}>Verifying session...</div>;
  }

  if (!user) {
    return <AuthScreen onLogin={setUser} />;
  }

  if (!workspace) {
    return (
      <RolePortal
        user={user}
        onOpenWorkspace={() => setWorkspace(true)}
        onLogout={logout}
      />
    );
  }

  return (
    <div>
      <div style={{
        padding: "10px 20px",
        background: "#f1f1fa",
        display: "flex",
        justifyContent: "space-between",
        alignItems: "center"
      }}>
        <span>Signed in: {user.name} ({user.role})</span>
        <button onClick={() => setWorkspace(false)}>Portal</button>
        <button onClick={logout}>Logout</button>
      </div>
      <App />
    </div>
  );
}

async function request(endpoint, options = {}) {
  const token = sessionStorage.getItem("urban_ripple_token");

  const response = await fetch(`${API_URL}${endpoint}`, {
    ...options,
    headers: {
      "Content-Type": "application/json",
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      ...options.headers,
    },
  });

  const data = await response.json().catch(() => ({}));

  if (!response.ok) {
    throw new Error(data.detail || `API error: ${response.status}`);
  }

  return data;
}

export async function getAnalyticsOverview(startDate = "", endDate = "") {
  const token = sessionStorage.getItem("urban_ripple_token");

  const params = new URLSearchParams();

  if (startDate) params.set("start_date", startDate);
  if (endDate) params.set("end_date", endDate);

  const query = params.toString();
  const url = `${API_URL}/api/analytics/overview${query ? `?${query}` : ""}`;

  const response = await fetch(url, {
    headers: {
      Authorization: `Bearer ${token}`
    }
  });

  if (!response.ok) {
    const result = await response.json().catch(() => ({}));
    throw new Error(result.detail || `Analytics request failed: ${response.status}`);
  }

  return response.json();
}
