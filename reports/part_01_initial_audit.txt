export async function registerUser(data) {
  const response = await fetch(`${API_URL}/api/auth/register`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },

const API_URL = import.meta.env.VITE_API_URL || "http://127.0.0.1:8000";

export async function getAnalyticsOverview(startDate = "", endDate = "") {
  const token = localStorage.getItem("urban_ripple_token");

  const params = new URLSearchParams();

  if (startDate) params.set("start_date", startDate);
  if (endDate) params.set("end_date", endDate);

  const query = params.toString();
  const url = `${API_URL}/api/analytics/overview${query ? `?${query}` : ""}`;

  const response = await fetch(url, {
    headers: {
      Authorization: `Bearer ${token}`
    }
  });

  if (!response.ok) {
    const result = await response.json().catch(() => ({}));
    throw new Error(result.detail || `Analytics request failed: ${response.status}`);
  }

  return response.json();
}

def require_roles(*allowed_roles):
    def dependency(user=Depends(get_current_user)):
        role = str(user.get("role", "public")).lower()

        if role not in allowed_roles:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Your role is not permitted to perform this action."
            )

        return user

    return dependency


def require_permission(permission):
    def dependency(user=Depends(get_current_user)):
        role = str(user.get("role", "public")).lower()

        return user

    return dependency
